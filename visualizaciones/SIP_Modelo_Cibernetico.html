// ══════════════════════════════════════════════════════
// SISTEMA INSTRUMENTAL PSICOTÉCNICO (SIP)
// Modelo Cibernético Formal
// Béguin & Rabardel (2000) · Kaptelinin (1996)
// ══════════════════════════════════════════════════════

SISTEMA SIP(sujeto, artefacto, contexto)

  // ── INICIALIZACIÓN ──────────────────────────────────
  ESTADO ← {
    vars_psicoemocionales: [estrés, autoeficacia,
                            pertenencia, motivación,
                            resiliencia, apoyo_social],
    esquema_utilización:   {reglas_operatorias,
                            invariantes_operatorias,
                            inferencias_posibles}
  }
  génesis_completa ← FALSO

══ BUCLE PRINCIPAL ═══════════════════════════════════
  MIENTRAS génesis_completa = FALSO:

    ① FEED-FORWARD  // anticipar ANTES del evento
       señal  ← ANTICIPAR(ESTADO.vars_psicoemocionales)
       riesgo ← ML_stacking.predecir(señal)
                 // RF + XGBoost + Gradient Boosting → FNN

    ② SELECCIÓN DE MEDIACIÓN INSTRUMENTAL
       SEGÚN tipo_mediación:
         PRAGMÁTICA  → ejecutar_acción(objetivo, contexto)
         EPISTÉMICA  → construir_conocimiento(señal, sujeto)
         HEURÍSTICA  → anticipar_obstáculo(riesgo, contexto)

    ③ INTERVENCIÓN
       respuesta ← intervenir(sujeto, tipo_mediación, riesgo)

    ④ FEEDBACK       // el sistema aprende del resultado
       ESTADO.vars_psicoemocionales ← ACTUALIZAR(respuesta)
       ESTADO.esquema_utilización   ← ACTUALIZAR(respuesta)
       artefacto                    ← AJUSTAR(respuesta)

    ⑤ CATACRESIS     // uso creativo no previsto
       SI catacresis_detectada(respuesta, diseño_original):
         artefacto ← REINSTRUMENTALIZAR(nueva_forma_uso)
         REGISTRAR principio_diseño_emergente()

    ⑥ EVALUACIÓN DE GÉNESIS
       SI madurez_instrumental(ESTADO, artefacto) ≥ umbral:
         génesis_completa ← VERDADERO

  FIN BUCLE ════════════════════════════════════════════

  // El SIP NUNCA termina realmente — génesis continua
  // Cada ciclo genera nuevo conocimiento transferible
  RETORNAR {
    nuevo_esquema_utilización,          // conocimiento acumulado
    principios_diseño_emergentes,       // → Artículo 2 (IBD)
    riesgo_abandono_predicho            // → Artículo 3 (ML)
  }
